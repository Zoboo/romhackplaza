cmake_minimum_required(VERSION 3.13)
project(romhackplaza LANGUAGES CXX)
option(ROMHACKPLAZA_BUILD_TESTS "Build offline module tests" OFF)
add_library(romhackplaza STATIC src/client.cpp patch.cpp)
target_include_directories(romhackplaza PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(romhackplaza PUBLIC cxx_std_11)
if(NINTENDO_SWITCH OR SWITCH)
 find_library(ROMHACKPLAZA_CURL_LIBRARY NAMES curl REQUIRED)
 find_library(ROMHACKPLAZA_Z_LIBRARY NAMES z REQUIRED)
 target_link_libraries(romhackplaza PUBLIC ${ROMHACKPLAZA_CURL_LIBRARY} ${ROMHACKPLAZA_Z_LIBRARY})
else()
 find_package(CURL REQUIRED)
 find_package(ZLIB REQUIRED)
 target_link_libraries(romhackplaza PUBLIC CURL::libcurl ZLIB::ZLIB)
endif()
if(ROMHACKPLAZA_BUILD_TESTS)
 enable_testing()
 add_executable(romhackplaza-client-tests tests/client-tests.cpp)
 target_link_libraries(romhackplaza-client-tests PRIVATE romhackplaza)
 add_test(NAME client-offline COMMAND romhackplaza-client-tests)
 add_executable(romhackplaza-patch-tests patch_tests.cpp)
 target_link_libraries(romhackplaza-patch-tests PRIVATE romhackplaza)
 add_test(NAME patch-offline COMMAND romhackplaza-patch-tests)
 add_executable(romhackplaza-rate-tests tests/rate-tests.cpp)
 target_include_directories(romhackplaza-rate-tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
 target_compile_features(romhackplaza-rate-tests PRIVATE cxx_std_11)
 if(NINTENDO_SWITCH OR SWITCH)
  target_link_libraries(romhackplaza-rate-tests PRIVATE ${ROMHACKPLAZA_CURL_LIBRARY} ${ROMHACKPLAZA_Z_LIBRARY})
 else()
  target_link_libraries(romhackplaza-rate-tests PRIVATE CURL::libcurl ZLIB::ZLIB)
 endif()
 add_test(NAME rate-offline COMMAND romhackplaza-rate-tests)
endif()
